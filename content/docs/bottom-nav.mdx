## Usage

```tsx
import Link from "next/link";
import { usePathname } from "next/navigation";
import { Home, Search, User } from "lucide-react";
import { BottomNav, BottomNavItem } from "@/components/ui/bottom-nav";

const pathname = usePathname();

<BottomNav value={pathname} linkComponent={Link} className="md:hidden">
  <BottomNavItem value="/" href="/" icon={<Home />}>Home</BottomNavItem>
  <BottomNavItem value="/search" href="/search" icon={<Search />}>Search</BottomNavItem>
  <BottomNavItem value="/profile" href="/profile" icon={<User />}>Profile</BottomNavItem>
</BottomNav>
```

The nav never reads the URL itself: `value` is whatever your router says is current. `linkComponent` is how a Next app keeps client-side routing (omit it and items are plain `<a>` tags). The component carries no breakpoint, so `md:hidden` is the layout's call. Re-tapping the active item scrolls back to the top.
