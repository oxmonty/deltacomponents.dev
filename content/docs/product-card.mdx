## Usage

```tsx
import {
  ProductCard,
  ProductCardContent,
  ProductCardHeader,
  ProductCardMetric,
  ProductCardSubtitle,
  ProductCardTitle,
} from "@/components/ui/product-card";

<ProductCard>
  <ProductCardContent>
    <ProductCardHeader>
      <ProductCardTitle>Twemco Clock</ProductCardTitle>
      <ProductCardSubtitle>Clock</ProductCardSubtitle>
    </ProductCardHeader>
    <ProductCardMetric>$219</ProductCardMetric>
  </ProductCardContent>
</ProductCard>
```
